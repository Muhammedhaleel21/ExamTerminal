//server.c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <arpa/inet.h>

#define PORT 8080
#define MAX 20

int main()
{
    int server_fd, new_socket;
    struct sockaddr_in address;
    int addrlen = sizeof(address);
    int r, c;
    int a[MAX][MAX], b[MAX][MAX], sum[MAX][MAX];

    server_fd = socket(AF_INET, SOCK_STREAM, 0);

    address.sin_family = AF_INET;
    address.sin_addr.s_addr = INADDR_ANY;
    address.sin_port = htons(PORT);

    bind(server_fd, (struct sockaddr *)&address, sizeof(address));
    listen(server_fd, 3);

    printf("Waiting for client connection...\n");

    new_socket = accept(server_fd, (struct sockaddr *)&address, (socklen_t *)&addrlen);

    read(new_socket, &r, sizeof(int));
    read(new_socket, &c, sizeof(int));

    read(new_socket, a, sizeof(a));
    read(new_socket, b, sizeof(b));

    printf("\nMatrix Addition:\n");

    for(int i = 0; i < r; i++)
    {
        for(int j = 0; j < c; j++)
        {
            sum[i][j] = a[i][j] + b[i][j];
            printf("%4d", sum[i][j]);
        }
        printf("\n");
    }

    write(new_socket, sum, sizeof(sum));

    printf("\nResult Sent to Client.\n");

    close(new_socket);
    close(server_fd);

    return 0;
}



//client.c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <arpa/inet.h>

#define PORT 8080
#define MAX 20

int main()
{
    int sock = 0;
    struct sockaddr_in serv_addr;
    int r, c;
    int a[MAX][MAX], b[MAX][MAX], sum[MAX][MAX];

    sock = socket(AF_INET, SOCK_STREAM, 0);

    serv_addr.sin_family = AF_INET;
    serv_addr.sin_port = htons(PORT);

    inet_pton(AF_INET, "127.0.0.1", &serv_addr.sin_addr);

    connect(sock, (struct sockaddr *)&serv_addr, sizeof(serv_addr));

    printf("Enter number of rows: ");
    scanf("%d", &r);

    printf("Enter number of columns: ");
    scanf("%d", &c);

    printf("\nEnter first matrix:\n");

    for(int i = 0; i < r; i++)
    {
        for(int j = 0; j < c; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("\nEnter second matrix:\n");

    for(int i = 0; i < r; i++)
    {
        for(int j = 0; j < c; j++)
        {
            scanf("%d", &b[i][j]);
        }
    }

    write(sock, &r, sizeof(int));
    write(sock, &c, sizeof(int));

    write(sock, a, sizeof(a));
    write(sock, b, sizeof(b));

    read(sock, sum, sizeof(sum));

    printf("\nResultant Matrix:\n");

    for(int i = 0; i < r; i++)
    {
        for(int j = 0; j < c; j++)
        {
            printf("%4d", sum[i][j]);
        }
        printf("\n");
    }

    close(sock);

    return 0;
}
