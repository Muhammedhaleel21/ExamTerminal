//server.c
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

#define PORT 8080
#define MAX 1024

int main()
{
    int sockfd, i, j, flag = 1;
    char str[MAX], result[50];

    struct sockaddr_in server_addr, client_addr;
    socklen_t len = sizeof(client_addr);

    sockfd = socket(AF_INET, SOCK_DGRAM, 0);

    server_addr.sin_family = AF_INET;
    server_addr.sin_addr.s_addr = INADDR_ANY;
    server_addr.sin_port = htons(PORT);

    bind(sockfd, (struct sockaddr *)&server_addr, sizeof(server_addr));

    printf("UDP Server Running\n");

    recvfrom(sockfd, str, MAX, 0,
             (struct sockaddr *)&client_addr, &len);

    str[strcspn(str, "\n")] = '\0';

    printf("Received string: %s\n", str);

    i = 0;
    j = strlen(str) - 1;

    while(i < j)
    {
        if(str[i] != str[j])
        {
            flag = 0;
            break;
        }

        i++;
        j--;
    }

    if(flag == 1)
        strcpy(result, "Palindrome");
    else
        strcpy(result, "Not Palindrome");

    sendto(sockfd, result, strlen(result) + 1, 0,
           (struct sockaddr *)&client_addr, len);

    printf("Result sent to client.\n");

    close(sockfd);

    return 0;
}



//client.c
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

#define PORT 8080
#define MAX 1024

int main()
{
    int sockfd;
    char str[MAX], buffer[50];

    struct sockaddr_in server_addr;
    socklen_t len = sizeof(server_addr);

    sockfd = socket(AF_INET, SOCK_DGRAM, 0);

    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(PORT);
    server_addr.sin_addr.s_addr = inet_addr("127.0.0.1");

    printf("Enter a string:\n");
    fgets(str, MAX, stdin);

    sendto(sockfd, str, strlen(str) + 1, 0,
           (struct sockaddr *)&server_addr, len);

    recvfrom(sockfd, buffer, sizeof(buffer), 0,
             (struct sockaddr *)&server_addr, &len);

    printf("\nResult: %s\n", buffer);

    close(sockfd);

    return 0;
}
