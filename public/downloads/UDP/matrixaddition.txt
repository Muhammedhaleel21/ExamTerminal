//server.c
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

#define PORT 8080
#define MAX 20

int main()
{
    int sockfd, r, c;
    int a[MAX][MAX], b[MAX][MAX], sum[MAX][MAX];

    struct sockaddr_in server_addr, client_addr;
    socklen_t len = sizeof(client_addr);

    sockfd = socket(AF_INET, SOCK_DGRAM, 0);

    server_addr.sin_family = AF_INET;
    server_addr.sin_addr.s_addr = INADDR_ANY;
    server_addr.sin_port = htons(PORT);

    bind(sockfd, (struct sockaddr *)&server_addr, sizeof(server_addr));

    printf("UDP Server Running\n");

    recvfrom(sockfd, &r, sizeof(r), 0,
             (struct sockaddr *)&client_addr, &len);

    recvfrom(sockfd, &c, sizeof(c), 0,
             (struct sockaddr *)&client_addr, &len);

    recvfrom(sockfd, a, r * c * sizeof(int), 0,
             (struct sockaddr *)&client_addr, &len);

    recvfrom(sockfd, b, r * c * sizeof(int), 0,
             (struct sockaddr *)&client_addr, &len);

    printf("Matrix Addition:\n");

    for(int i = 0; i < r; i++)
    {
        for(int j = 0; j < c; j++)
        {
            sum[i][j] = a[i][j] + b[i][j];
            printf("%d ", sum[i][j]);
        }
        printf("\n");
    }

    sendto(sockfd, sum, r * c * sizeof(int), 0,
           (struct sockaddr *)&client_addr, len);

    printf("Result sent to client.\n");

    close(sockfd);

    return 0;
}


//client.c

#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

#define PORT 8080
#define MAX 20

int main()
{
    int sockfd, r, c;
    int a[MAX][MAX], b[MAX][MAX], sum[MAX][MAX];

    struct sockaddr_in server_addr;
    socklen_t len = sizeof(server_addr);

    sockfd = socket(AF_INET, SOCK_DGRAM, 0);

    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(PORT);
    server_addr.sin_addr.s_addr = inet_addr("127.0.0.1");

    printf("Enter rows and columns:\n");
    scanf("%d%d", &r, &c);

    printf("Enter Matrix A:\n");

    for(int i = 0; i < r; i++)
    {
        for(int j = 0; j < c; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Enter Matrix B:\n");

    for(int i = 0; i < r; i++)
    {
        for(int j = 0; j < c; j++)
        {
            scanf("%d", &b[i][j]);
        }
    }

    sendto(sockfd, &r, sizeof(r), 0,
           (struct sockaddr *)&server_addr, len);

    sendto(sockfd, &c, sizeof(c), 0,
           (struct sockaddr *)&server_addr, len);

    sendto(sockfd, a, r * c * sizeof(int), 0,
           (struct sockaddr *)&server_addr, len);

    sendto(sockfd, b, r * c * sizeof(int), 0,
           (struct sockaddr *)&server_addr, len);

    recvfrom(sockfd, sum, r * c * sizeof(int), 0,
             (struct sockaddr *)&server_addr, &len);

    printf("\nResultant Matrix:\n");

    for(int i = 0; i < r; i++)
    {
        for(int j = 0; j < c; j++)
        {
            printf("%d\t", sum[i][j]);
        }
        printf("\n");
    }

    close(sockfd);

    return 0;
}
